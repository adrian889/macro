function doGet(e) { // 1. Dacă cererea vine din GitHub Pages (fetch cu ?action=load), returnăm JSON! if (e && e.parameter && e.parameter.action === 'load') { return jsonResponse(getAllSheetData()); } // 2. Altfel, dacă deschizi linkul direct în browser, afișăm aplicația HTML return HtmlService.createHtmlOutputFromFile('Index') .setTitle('Macro Tracker RO') .addMetaTag('viewport', 'width=device-width, initial-scale=1, maximum-scale=1, user-scalable=no') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } function doPost(e) { try { var raw = (e && e.postData && e.postData.contents) ? e.postData.contents : '{}'; var payload = JSON.parse(raw); if (payload.action === 'syncAll') { return jsonResponse(syncAllFromClient(payload)); } return jsonResponse({ status: 'error', message: 'Actiune necunoscuta' }); } catch (err) { return jsonResponse({ status: 'error', message: err.toString() }); } } function syncAllFromClient(payload) { var ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. Sincronizare sigură Tracker (cu păstrarea formulei în coloana F) if (payload.entries && Array.isArray(payload.entries)) { var tSheet = getOrCreateTrackerSheet(ss); var rows = [['Data', 'Meal/Item', 'Protein (g)', 'Fat (g)', 'Carbs (g)', 'Calories (kcal)']]; payload.entries.forEach(function(r, idx) { var rowNum = idx + 2; // Rândul 1 este capul de tabel var p = Number(r.p) || 0; var f = Number(r.f) || 0; var c = Number(r.c) || 0; // REPARAT AICI: Am scos ", 0" ca să meargă și pe Google Sheets în limba română var kcalFormula = '=ROUND(C' + rowNum + '*4 + D' + rowNum + '*9 + E' + rowNum + '*4; 0)'; rows.push([r.date, r.name, p, f, c, kcalFormula]); }); var lastRow = tSheet.getLastRow(); tSheet.getRange(1, 1, rows.length, 6).setValues(rows); if (lastRow > rows.length) { tSheet.getRange(rows.length + 1, 1, lastRow - rows.length, 6).clearContent(); } // Extindem automat foaia "Zile" cu datele noi și copiem formulele în regim batch syncZileSheetDates(ss, payload.entries); } // 2. Sincronizare sigură Bancă Alimente (Porții fixe) if (payload.foodBank && Array.isArray(payload.foodBank)) { var bSheet = getOrCreateBankSheet(ss); var bRows = [['ID', 'Denumire', 'Categorie', 'Protein (g)', 'Fat (g)', 'Carbs (g)', 'Frecventa']]; payload.foodBank.forEach(function(b) { bRows.push([ b.id, b.name, b.cat || 'Mese Gătite', Number(b.p) || 0, Number(b.f) || 0, Number(b.c) || 0, Number(b.freq) || 1 ]); }); var bLastRow = bSheet.getLastRow(); bSheet.getRange(1, 1, bRows.length, 7).setValues(bRows); if (bLastRow > bRows.length) { bSheet.getRange(bRows.length + 1, 1, bLastRow - bRows.length, 7).clearContent(); } } // 3. Sincronizare sigură Alimente_100g (Calculator la 100g) if (payload.foods100g && Array.isArray(payload.foods100g)) { var fSheet = getOrCreateFoods100Sheet(ss); var fRows = [['ID', 'Denumire', 'Categorie', 'Protein (g)', 'Fat (g)', 'Carbs (g)', 'Frecventa']]; payload.foods100g.forEach(function(item) { fRows.push([ item.id, item.name, item.cat || 'Mese Gătite', Number(item.p) || 0, Number(item.f) || 0, Number(item.c) || 0, Number(item.freq) || 1 ]); }); var fLastRow = fSheet.getLastRow(); fSheet.getRange(1, 1, fRows.length, 7).setValues(fRows); if (fLastRow > fRows.length) { fSheet.getRange(fRows.length + 1, 1, fLastRow - fRows.length, 7).clearContent(); } } // 4. Sincronizare Profil DIRECT în foaia "Date" (Batch pe celulele de input) if (payload.profile) { writeProfileToDateSheet(ss, payload.profile); } SpreadsheetApp.flush(); return { status: 'ok', mode: 'syncAll', updatedAt: new Date().toISOString() }; } /** * Adaugă automat zilele noi în foaia "Zile" și copiază formulele într-o singură operație batch */ function syncZileSheetDates(ss, entries) { var zSheet = ss.getSheetByName('Zile'); if (!zSheet) return; var lastRow = zSheet.getLastRow(); if (lastRow < 2) return; var existingVals = zSheet.getRange(1, 1, lastRow, 1).getValues(); var existingDates = {}; var maxDateStr = ''; var lastValidDateRow = -1; for (var i = 0; i < existingVals.length; i++) { if (!existingVals[i][0]) continue; var dStr = formatSheetDate(existingVals[i][0]); if (/^\d{4}-\d{2}-\d{2}$/.test(dStr)) { existingDates[dStr] = true; lastValidDateRow = i + 1; // Index 1-based al ultimului rând cu dată validă if (dStr > maxDateStr) maxDateStr = dStr; } } if (lastValidDateRow === -1) return; // Colectăm toate datele din Tracker care sunt mai recente decât ultima dată din "Zile" var newDatesMap = {}; entries.forEach(function(e) { var d = String(e.date || '').trim().slice(0, 10); if (/^\d{4}-\d{2}-\d{2}$/.test(d) && !existingDates[d] && d > maxDateStr) { newDatesMap[d] = true; } }); var newDates = Object.keys(newDatesMap).sort(); if (newDates.length === 0) return; var numNew = newDates.length; var maxCols = Math.max(8, zSheet.getLastColumn()); var startRow = lastValidDateRow + 1; // Dacă există rânduri de sinteză sub ultimul rând cu dată, inserăm rânduri noi ca să nu le suprascriem if (lastValidDateRow < lastRow) { zSheet.insertRowsAfter(lastValidDateRow, numNew); } var sourceRowRange = zSheet.getRange(lastValidDateRow, 1, 1, maxCols); var targetRowsRange = zSheet.getRange(startRow, 1, numNew, maxCols); // 1. Copiem formatarea pe toate rândurile noi dintr-un singur apel sourceRowRange.copyTo( targetRowsRange, SpreadsheetApp.CopyPasteType.PASTE_FORMAT, false ); // 2. Copiem formulele din coloanele B..H pe toate rândurile noi dintr-un singur apel zSheet.getRange(lastValidDateRow, 2, 1, maxCols - 1).copyTo( zSheet.getRange(startRow, 2, numNew, maxCols - 1), SpreadsheetApp.CopyPasteType.PASTE_FORMULA, false ); // 3. Scriem toate datele noi în coloana A printr-un singur setValues() var dateCells = newDates.map(function(d) { return [d]; }); zSheet.getRange(startRow, 1, numNew, 1).setValues(dateCells); } function getAllSheetData() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var tSheet = getOrCreateTrackerSheet(ss); var tData = tSheet.getDataRange().getValues(); var entries = []; for (var i = 1; i < tData.length; i++) { var row = tData[i]; if (!row[0] && !row[1]) continue; var dateFormatted = formatSheetDate(row[0]); entries.push({ id: 't_' + i + '_' + dateFormatted, date: dateFormatted, name: String(row[1] || '').trim(), p: parseNum(row[2]), f: parseNum(row[3]), c: parseNum(row[4]) }); } var bSheet = ss.getSheetByName('Banca_Alimente'); var foodBank = null; if (bSheet && bSheet.getLastRow() > 1) { var bData = bSheet.getDataRange().getValues(); foodBank = []; for (var j = 1; j < bData.length; j++) { var br = bData[j]; if (!br[1]) continue; foodBank.push({ id: String(br[0] || ('b_' + j)), name: String(br[1]).trim(), cat: String(br[2] || 'Mese Gătite'), p: parseNum(br[3]), f: parseNum(br[4]), c: parseNum(br[5]), freq: parseNum(br[6]) || 1 }); } } var fSheet = ss.getSheetByName('Alimente_100g'); var foods100g = []; if (fSheet && fSheet.getLastRow() > 1) { var fData = fSheet.getDataRange().getValues(); for (var k = 1; k < fData.length; k++) { var fr = fData[k]; if (!fr[1]) continue; foods100g.push({ id: String(fr[0] || ('f100_' + k)), name: String(fr[1]).trim(), cat: String(fr[2] || 'Mese Gătite'), p: parseNum(fr[3]), f: parseNum(fr[4]), c: parseNum(fr[5]), freq: parseNum(fr[6]) || 1 }); } } var profile = readProfileFromDateSheet(ss); return { status: 'ok', entries: entries, foodBank: foodBank, foods100g: foods100g, profile: profile }; } function readProfileFromDateSheet(ss) { var dSheet = ss.getSheetByName('Date'); if (!dSheet) return null; var vals = dSheet.getRange(1, 1, 21, 4).getValues(); var age = parseNum(vals[2][1]) || 36; // B3: Vârsta var height = parseNum(vals[3][1]) || 179; // B4: Înălțime var weight = parseNum(vals[4][1]) || 78.5; // B5: Greutate Actuală var neck = parseNum(vals[7][1]) || 37; // B8: Circumferință Gât var waist = parseNum(vals[8][1]) || 86; // B9: Circumferință Talie var actStr = String(vals[13][1] || ''); var actMatch = actStr.match(/(\d+[.,]\d+)/); var activityFactor = actMatch ? parseNum(actMatch[1]) : 1.40; var adjRaw = String(vals[15][1] || '').toLowerCase(); var calorieAdj = -300; if (adjRaw.indexOf('men') !== -1 || adjRaw === '0') { calorieAdj = 0; } else if (adjRaw.indexOf('surplus') !== -1) { calorieAdj = 300; } else if (!isNaN(parseFloat(adjRaw))) { calorieAdj = parseNum(adjRaw); } var targetKcal = parseNum(vals[17][1]) || 1891; // B18 var targetP = parseNum(vals[18][1]) || 141.3; // B19 var targetF = parseNum(vals[19][1]) || 62.8; // B20 var targetC = parseNum(vals[20][1]) || 190; // B21 var proteinPerKg = weight > 0 ? Number((targetP / weight).toFixed(2)) : 1.8; var fatPerKg = weight > 0 ? Number((targetF / weight).toFixed(2)) : 0.8; var dexaNote = String(vals[9][3] || ''); var dexaMatch = dexaNote.match(/([+-]?\d+[.,]\d+)\s*%\s*\)/); var dexaOffset = dexaMatch ? parseNum(dexaMatch[1]) : 11.7; return { age: age, height: height, weight: weight, neck: neck, waist: waist, dexaOffset: dexaOffset, activityFactor: activityFactor, calorieAdj: calorieAdj, proteinPerKg: proteinPerKg, fatPerKg: fatPerKg }; } function writeProfileToDateSheet(ss, p) { var dSheet = ss.getSheetByName('Date'); if (!dSheet) return; // 1. Actualizare batch pentru B3:B5 (Vârstă, Înălțime, Greutate) fără a atinge B6 (BMI) și B7 (FFMI) var rangeB3B5 = dSheet.getRange('B3:B5'); var valsB3B5 = rangeB3B5.getValues(); var b3b5Changed = false; if (p.age && parseNum(valsB3B5[0][0]) !== Number(p.age)) { valsB3B5[0][0] = Number(p.age); b3b5Changed = true; } if (p.height && parseNum(valsB3B5[1][0]) !== Number(p.height)) { valsB3B5[1][0] = Number(p.height); b3b5Changed = true; } if (p.weight && parseNum(valsB3B5[2][0]) !== Number(p.weight)) { valsB3B5[2][0] = Number(p.weight); b3b5Changed = true; } if (b3b5Changed) { rangeB3B5.setValues(valsB3B5); } // 2. Actualizare batch pentru B8:D9 (Gât, Talie și data modificării în coloana D) var rangeNeckWaist = dSheet.getRange('B8:D9'); var nwVals = rangeNeckWaist.getValues(); var todayFormatted = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd'); var nwChanged = false; if (p.neck && parseNum(nwVals[0][0]) !== Number(p.neck)) { nwVals[0][0] = Number(p.neck); nwVals[0][2] = todayFormatted; nwChanged = true; } if (p.waist && parseNum(nwVals[1][0]) !== Number(p.waist)) { nwVals[1][0] = Number(p.waist); nwVals[1][2] = todayFormatted; nwChanged = true; } if (nwChanged) { rangeNeckWaist.setValues(nwVals); } } function getOrCreateTrackerSheet(ss) { var s = ss.getSheetByName('Tracker'); if (!s) { s = ss.insertSheet('Tracker'); s.appendRow(['Data', 'Meal/Item', 'Protein (g)', 'Fat (g)', 'Carbs (g)', 'Calories (kcal)']); } return s; } function getOrCreateBankSheet(ss) { var s = ss.getSheetByName('Banca_Alimente'); if (!s) { s = ss.insertSheet('Banca_Alimente'); s.appendRow(['ID', 'Denumire', 'Categorie', 'Protein (g)', 'Fat (g)', 'Carbs (g)', 'Frecventa']); } return s; } function getOrCreateFoods100Sheet(ss) { var s = ss.getSheetByName('Alimente_100g'); if (!s) { s = ss.insertSheet('Alimente_100g'); s.appendRow(['ID', 'Denumire', 'Categorie', 'Protein (g)', 'Fat (g)', 'Carbs (g)', 'Frecventa']); } return s; } function formatSheetDate(val) { if (Object.prototype.toString.call(val) === '[object Date]') { return Utilities.formatDate(val, Session.getScriptTimeZone(), 'yyyy-MM-dd'); } return String(val).trim().slice(0, 10); } function parseNum(v) { if (typeof v === 'number') return Number(v.toFixed(2)); if (!v) return 0; var n = parseFloat(String(v).replace(',', '.').trim()); return isNaN(n) ? 0 : Number(n.toFixed(2)); } function jsonResponse(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
